CREATE TABLE DWD.sjzl_asset_unified_alert_event
(
    `soc_uuid` String DEFAULT generateUUIDv4(),
    `soc_date` Date DEFAULT CAST(now(), 'Date'),
    `alert_type` String COMMENT '告警分类编码，详见附录A',
    `alert_name` String COMMENT '告警名称',
    `alert_id` UInt64 COMMENT '告警规则ID',
    `event_id` UInt64 COMMENT '事件标识',
    `alert_level` String COMMENT '告警等级：1-低 2-中 3-高',
    `vendor_code` String COMMENT '厂商标识，如Lvmeng',
    `monitor_time` DateTime COMMENT '告警发现时间，精确到秒',
    `device_ip` String COMMENT '设备IP',
    `device_id` String COMMENT '设备详细ID信息',
    `device_name` String COMMENT '设备具体名称',
    `sip_ipv4` String COMMENT '源IPv4',
    `sip_ipv6` String COMMENT '源IPv6',
    `sport` UInt16 COMMENT '传输层源端口',
    `dip_ipv4` String COMMENT '目的IPv4',
    `dip_ipv6` String COMMENT '目的IPv6',
    `dport` UInt16 COMMENT '传输层目的端口',
    `protocol` String COMMENT '传输层协议 TCP/UDP/Http等',
    `app_protocol` String COMMENT '应用层协议 HTTP/TELNET等',
    `host` String COMMENT '受影响域名',
    `url` String COMMENT '受影响URL',
    `method` String COMMENT 'HTTP请求方法',
    `path` String COMMENT '受影响资源路径',
    `response_code` UInt32 COMMENT '协议响应码',
    `request_body` String COMMENT '请求body，base64编码',
    `response_body` String COMMENT '响应body，base64编码',
    `rawinfo` String COMMENT '命中告警的payload，base64编码',
    `file_name` String COMMENT '恶意文件名称',
    `file_md5` String COMMENT '文件MD5',
    `metadata_id` String COMMENT '关联会话唯一标识',
    `attack_direction` String COMMENT '攻击方向 1内到外 2外到内 3内到内 4其他 0未知',
    `attack_result` String COMMENT '攻击结果 0/1尝试 3成功 6失败',
    `smac` String COMMENT '源MAC',
    `dmac` String COMMENT '目的MAC',
    `is_pcap` String COMMENT '存包标志 1可下载 0不可下载',
    `wangan_ruleid` UInt32 COMMENT '网安规则ID',
    `com_code` String,
    `dev_ip` String,
    `json_mess` String,
    `version` String COMMENT '版本号',
    `soc_date_time` DateTime DEFAULT now() COMMENT '日期时间',
    `soc_date_time_timestamp` UInt64 DEFAULT toUnixTimestamp(toDateTime64(soc_date_time, 3)) * 1000 COMMENT '日期时间-时间戳',
    `report_time` DateTime DEFAULT now() COMMENT '日志时间',
    `report_time_timestamp` UInt64 DEFAULT toUnixTimestamp(toDateTime64(report_time, 3)) * 1000 COMMENT '日志时间-时间戳',
    `orig_log` Nullable(String) DEFAULT NULL COMMENT '原始日志',
    `log_type` String COMMENT '日志类型',
    `login_name` String COMMENT '登录姓名'
)
ENGINE = MergeTree
PARTITION BY toDate(monitor_time)
ORDER BY (monitor_time, device_ip, alert_id)
SETTINGS index_granularity = 8192
COMMENT '网络安全告警事件数据表'
