CREATE MATERIALIZED VIEW DWD.zjg_traffic_app_audit_mv TO DWD.zjg_traffic_app_audit
(
    `updated_at` UInt64,
    `updated_at_date_time` DateTime('Asia/Shanghai'),
    `updated_at_date` Date,
    `dev_id` String,
    `data_type` UInt8,
    `app_type` UInt8,
    `app_middleware` Nullable(String),
    `middleware_version` Nullable(String),
    `ip` Nullable(String),
    `port` UInt32,
    `extended_fields` Nullable(String)
)
AS SELECT
    updated_at,
    toDateTime(updated_at / 1000, 'Asia/Shanghai') AS updated_at_date_time,
    toDate(toDateTime(updated_at / 1000, 'Asia/Shanghai')) AS updated_at_date,
    device_id AS dev_id,
    data_type,
    app_type,
    app_middleware,
    middleware_version,
    ip,
    port,
    extended_fields
FROM DWD.zjg_traffic_app_audit_kafka
