CREATE MATERIALIZED VIEW DWD.mv_ods_to_dwd_zjsj_softchange TO DWD.sjzl_asset_zjg_host_audit
(
    `company` String,
    `product` String,
    `host_name` String,
    `user_name` String,
    `department` String,
    `department_id` String,
    `full_department` String,
    `host_ip` String,
    `host_mac` String,
    `unit` String,
    `desc` String,
    `host_id` String,
    `action_type` String,
    `login_account_name` String,
    `login_role_type` String,
    `user_id` String,
    `send_time` String,
    `log_time` String,
    `critical_level` String,
    `behavior_type` String,
    `handle` String,
    `result` String,
    `log_event` String
)
AS SELECT
    '北信源' AS company,
    '安全套件' AS product,
    device_name AS host_name,
    registrant_name AS user_name,
    device_department_name AS department,
    device_department_id AS department_id,
    device_department_full_name AS full_department,
    ip AS host_ip,
    mac AS host_mac,
    company_name AS unit,
    description AS desc,
    device_id AS host_id,
    action AS action_type,
    os_login AS login_account_name,
    event_type AS login_role_type,
    registrant_id AS user_id,
    send_time AS send_time,
    log_time AS log_time,
    risk_level AS critical_level,
    event_class AS behavior_type,
    handle,
    result,
    '硬件变更日志' AS log_event
FROM ODS.sjzl_asset_taojian
WHERE log_type = 25
