CREATE TABLE DWD.sjzl_flow_communication
(
    `soc_uuid` String DEFAULT generateUUIDv4(),
    `tenant_id` Nullable(String),
    `soc_date_time` DateTime DEFAULT now(),
    `soc_date` Date MATERIALIZED toDate(soc_date_time),
    `soc_date_time_timestamp` UInt64 MATERIALIZED toUnixTimestamp(soc_date_time) * 1000 COMMENT '创建时间时间戳',
    `device_id` String COMMENT '设备ID',
    `device_port_id` Int32 COMMENT '设备接口ID',
    `device_ip` String COMMENT '设备IP',
    `interface_icon` String COMMENT '接口标识',
    `data_type` Int32 COMMENT '数据类型',
    `time` UInt64 COMMENT '时间，单位毫秒',
    `event_date_time` DateTime DEFAULT toDateTime(time / 1000, 'Asia/Shanghai'),
    `event_date` Date DEFAULT toDate(toDateTime(time / 1000)),
    `sip` String COMMENT '源IPv4地址',
    `src_com_code` String,
    `sipv6` String COMMENT '源IPv6地址',
    `smac` String COMMENT '源MAC地址',
    `sport` Int32 COMMENT '源端口',
    `dip` String COMMENT '目的IPv4',
    `dst_com_code` String,
    `dipv6` String COMMENT '目的IPv6 地址',
    `dmac` String COMMENT '目的MAC地址',
    `dport` Int32 COMMENT '目的端口',
    `network_protocol` Int32 COMMENT '网络层协议',
    `transport_protocol` Int32 COMMENT '传输层协议',
    `session_protocol` Int32 COMMENT '会话层协议',
    `app_protocol` Int32 COMMENT '应用层协议',
    `sess_id` String COMMENT '会话ID',
    `total_in_bytes` Int32 COMMENT '流入数据总字节',
    `total_out_bytes` Int32 COMMENT '流出数据总字节',
    `flow_start_time` Int64 COMMENT '流开始时间',
    `flow_end_time` Int64 COMMENT '流结束时间',
    `flow_duration` Int32 COMMENT '流持续时间',
    `tcp_flag` Int32 COMMENT 'TCP流标记',
    `close_status` String COMMENT 'TCP数据流的结束方式',
    `log_type` Int32 COMMENT '日志类型',
    `user_name` String COMMENT '用户名',
    `res_code` Int32 COMMENT '登录结果码，包括：0.失败，l.成功，2.其他',
    `extended_fields` Nullable(String) COMMENT '扩展字段集合',
    `is_extend_log` UInt8 DEFAULT 0 COMMENT '是否是扩展日志，默认0-否 1-是',
    `event_source` String COMMENT '数据来源',
    `orig_pkts` String COMMENT '会话源发送包数量',
    `resp_pkts` String COMMENT '返回包数',
    `packets_ts` String COMMENT '数据包时间戳',
    `packets_size` String COMMENT '数据包大小',
    `packets_dir` String COMMENT '数据包传输方向',
    `packets_ip_flags` String COMMENT '数据包IP标志位',
    `packets_ip_ttl` String COMMENT '数据包IP层存活时间',
    `tcp_header_len` String COMMENT '每个数据包包头长度（以会话为单位）（TCP）',
    `fwd_packet_length_min` String COMMENT '正向所有数据包的最小完整包长',
    `fwd_packet_length_mean` String COMMENT '所有正向数据包的完整包长平均值',
    `bwd_packet_length_min` String COMMENT '反向所有数据包的最小完整包长',
    `flow_iat_min` String COMMENT '所有相邻数据包达到间隔的最小时间',
    `flow_iat_std` String COMMENT '所有相邻数据包达到间隔的标准差',
    `bwd_iat_min` String COMMENT '反向所有数据包中相邻两个回包之间间隔的最小时间',
    `bwd_packets_s` String COMMENT '反向数据包速率',
    `fwd_header_length` String COMMENT '正向单个数据包的IP+TCP头部总长度',
    `bwd_header_length` String COMMENT '反向单个数据包的IP+TCP头部总长度',
    `psh_count` String COMMENT '携带PSH标志位的数据包总个数',
    `urg_count` String COMMENT '携带URG标志位的数据包总个数',
    `fwd_packet_length_max` String COMMENT '正向所有数据包的最大完整包长',
    `bytecount` String COMMENT '整条流量会话传输的全部字节总量',
    `com_code` String COMMENT '单位编码'
)
ENGINE = MergeTree
PARTITION BY toYYYYMMDD(soc_date)
ORDER BY (time, data_type, log_type)
TTL soc_date + toIntervalDay(180)
SETTINGS index_granularity = 8192
COMMENT '保密自监管-网络通联行为审计'
