CREATE MATERIALIZED VIEW ODS.sjzl_asset_ruixing_ESM_raw_mv TO DWD.sjzl_asset_ruixing_ESM_raw
(
    `soc_uuid` String,
    `soc_date` Date,
    `soc_date_time` DateTime,
    `soc_date_time_timestamp` UInt64,
    `report_time` DateTime,
    `report_time_timestamp` UInt64,
    `com_code` String,
    `key1` String,
    `taskname` String,
    `virusname` String,
    `filepath` String,
    `id` String,
    `state` String,
    `ip` String,
    `threatmethod` String,
    `virusclass` String,
    `leix` String,
    `time` DateTime,
    `table` String,
    `timestamp` String,
    `dev_ip` String,
    `flag` UInt8,
    `uuid` UUID,
    `json_mess` Nullable(String),
    `user_account` String,
    `recep_time` UInt16,
    `org_address_code` String,
    `virus_type` String
)
AS SELECT
    soc_uuid,
    toDate(time) AS soc_date,
    time AS soc_date_time,
    toUnixTimestamp(toDateTime64(time, 3)) * 1000 AS soc_date_time_timestamp,
    time AS report_time,
    toUnixTimestamp(toDateTime64(time, 3)) * 1000 AS report_time_timestamp,
    com_code,
    key1,
    taskname,
    virusname,
    filepath,
    id,
    state,
    ip,
    threatmethod,
    virusclass,
    leix,
    time,
    `table`,
    timestamp,
    dev_ip,
    flag,
    uuid,
    json_mess,
    user_account,
    recep_time,
    org_address_code,
    virus_type
FROM ODS.sjzl_asset_ruixing_ESM_raw
